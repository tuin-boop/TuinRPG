ACTOR BloxPlasmaRifle : PlasmaRifle replaces PlasmaRifle
{
	Weapon.SlotNumber 6
	Weapon.BobSpeed 1.5
	Weapon.BobRangeX 0.75
	Weapon.BobRangeY 0.75
	Weapon.BobStyle "InverseSmooth"
   	States
   	{
	Spawn:
		TNT1 A 0 nodelay A_JumpIf(CallACS("skintoggler")==1, "64spawn")		
		PLAS A 5
		loop
	Ready:
		TNT1 A 0 A_JumpIf(CallACS("skintoggler")==1, "64ready")
		PLRL A 1 A_WeaponReady
		Loop
	Deselect:
		TNT1 A 0 A_JumpIf(health <= 0, "DeadDeselect")
		TNT1 A 0 A_JumpIf(CallACS("skintoggler")==1, "64deselect")
		TNT1 A 0 a_jumpif(callacs("skintoggler")==2, "bldeselect")
		PLGS ABCDEFGHIKJLMMNN 1 A_Lower
		Loop
		
	BLDeselect:
		TNT1 A 0 A_JumpIf(health <= 0, "DeadBLDeselect")
		TNT1 A 0 A_JumpIf(CallACS("skintoggler")==1, "64deselect")
		BKPS ABCDEFGHIKJLMMNN 1 A_Lower
		Loop
		
	DeadDeselect:
		TNT1 A 0 A_JumpIf(CallACS("SkinToggler")==2, "DeadBLdeselect")
		TNT1 A 0 A_JumpIfInventory("GibToken", 1, "Deathgibbed")
		TNT1 A 0 A_JumpIfInventory("DeathToken", 1, "DeathState")
		PLRL A 1 A_Lower
		Loop
		
	DeadBLDeselect:
		TNT1 A 0 A_JumpIfInventory("GibToken", 1, "Deathgibbed")
		TNT1 A 0 A_JumpIfInventory("DeathToken", 1, "BLDeathState")
		PLRL A 1 A_Lower
		Loop
		
	Select:
		TNT1 A 0 A_JumpIf(CallACS("skintoggler")==2, "BLSelect")
		TNT1 A 0 a_jumpif(callacs("WeaponSoundToggle")==1, "silentselect")
		TNT1 A 0 a_jumpif(callacs("WeaponSoundToggle")==3, "silentselect")
		TNT1 A 0 A_JumpIf(CallACS("skintoggler")==1, "64select")
		TNT1 A 0 A_PlaySound("weapons/plasma/draw")
		PLGS NNMMLKJIHGFEDCBA 1 A_Raise
		goto ready
		
	SilentSelect:
		TNT1 A 0 a_jumpif(callacs("WeaponSoundToggle")==0, "select")
		TNT1 A 0 a_jumpif(callacs("WeaponSoundToggle")==2, "select")
		TNT1 A 0 A_JumpIf(CallACS("skintoggler")==1, "64select")
		PLGS NNMMLKJIHGFEDCBA 1 A_Raise
		goto ready
		
	BLSelect:
		TNT1 A 0 a_jumpif(callacs("WeaponSoundToggle")==1, "BLsilentselect")
		TNT1 A 0 a_jumpif(callacs("WeaponSoundToggle")==3, "BLsilentselect")
		TNT1 A 0 A_JumpIf(CallACS("skintoggler")==1, "64select")
		TNT1 A 0 A_PlaySound("weapons/plasma/draw")
		BKPS NNMMLKJIHGFEDCBA 1 A_Raise
		goto ready
		
	BLSilentSelect:
		TNT1 A 0 a_jumpif(callacs("WeaponSoundToggle")==0, "select")
		TNT1 A 0 a_jumpif(callacs("WeaponSoundToggle")==2, "select")
		TNT1 A 0 A_JumpIf(CallACS("skintoggler")==1, "64select")
		BKPS NNMMLKJIHGFEDCBA 1 A_Raise
		goto ready
		
	Fire:
		TNT1 A 0 A_JumpIf(CallACS("skintoggler")==1, "64fire")
		TNT1 A 0 a_jumpif(callacs("BobToggle")==1, "bobfire")
		
		PLRL B 1 Bright A_FirePlasma
					TNT1 A 0 A_JumpIf(CallACS("recoiltoggle")==0, "+2")
					TNT1 a 0 a_quakeex(1,1,1,2,0,10,"misc/null",QF_SCALEDOWN)		
		
		PLRL C 1 Bright
		PLRL D 1 Bright
		TNT1 A 0 A_ReFire(+1)
			goto cooldown
		PLRL C 1 Bright A_FirePlasma
					TNT1 A 0 A_JumpIf(CallACS("recoiltoggle")==0, "+2")
					TNT1 a 0 a_quakeex(1,1,1,2,0,10,"misc/null",QF_SCALEDOWN)		
	
		PLRL D 1 Bright
		PLRL E 1 Bright 
		PLRL A 0 a_refire(+1)
			goto cooldown
		PLRL B 1 Bright A_FirePlasma
					TNT1 A 0 A_JumpIf(CallACS("recoiltoggle")==0, "+2")
					TNT1 a 0 a_quakeex(1,1,1,2,0,10,"misc/null",QF_SCALEDOWN)		
		
		PLRL C 1 Bright
		PLRL D 1 Bright 

		PLRL A 0 a_refire("fire")
			goto cooldown
	bobFire:
		TNT1 A 0 A_JumpIf(CallACS("recoiltoggle")==1, "recoilbobfire")
		
		TNT1 A 0 A_WeaponReady(WRF_NoFire|WRF_NoSwitch)
		PLRL B 1 Bright A_FirePlasma
					TNT1 A 0 A_JumpIf(CallACS("recoiltoggle")==0, "+2")
					TNT1 a 0 a_quakeex(1,1,1,2,0,10,"misc/null",QF_SCALEDOWN)		

		TNT1 A 0 A_WeaponReady(WRF_NoFire|WRF_NoSwitch)		
		PLRL C 1 Bright
		TNT1 A 0 A_WeaponReady(WRF_NoFire|WRF_NoSwitch)
		PLRL D 1 Bright
		TNT1 A 0 A_ReFire //(+1)
			goto cooldown
		PLRL C 1 Bright A_FirePlasma	
					TNT1 A 0 A_JumpIf(CallACS("recoiltoggle")==0, "+2")
					TNT1 a 0 a_quakeex(1,1,1,2,0,10,"misc/null",QF_SCALEDOWN)		
		
		TNT1 A 0 A_WeaponReady(WRF_NoFire|WRF_NoSwitch)
		PLRL D 1 Bright
		TNT1 A 0 A_WeaponReady(WRF_NoFire|WRF_NoSwitch)
		PLRL E 1 Bright 
		PLRL A 0 a_refire //(+1)
			goto cooldown
		TNT1 A 0 A_WeaponReady(WRF_NoFire|WRF_NoSwitch)
		PLRL B 1 Bright A_FirePlasma
					TNT1 A 0 A_JumpIf(CallACS("recoiltoggle")==0, "+2")
					TNT1 a 0 a_quakeex(1,1,1,2,0,10,"misc/null",QF_SCALEDOWN)		
		
		TNT1 A 0 A_WeaponReady(WRF_NoFire|WRF_NoSwitch)
		PLRL C 1 Bright
		TNT1 A 0 A_WeaponReady(WRF_NoFire|WRF_NoSwitch)
		PLRL D 1 Bright 
		PLRL A 0 a_refire("fire")
			goto cooldown
		
		
	cooldown:
		TNT1 A 0 A_JumpIf(CallACS("skintoggler")==2, "BLcooldown")
		PLRL F 1  a_zoomfactor(1.0, ZOOM_NOSCALETURNING)
		PLRL GHI 1
		PLRL JKLM 2
		PLRL M 1
		PLRL N 2
		PLRL MOPQ 1
		PLRL R 1
		Goto Ready
	BLcooldown:
		BKPL F 1  a_zoomfactor(1.0, ZOOM_NOSCALETURNING)
		BKPL GHI 1
		BKPL JKLM 2
		BKPL M 1
		BKPL N 2
		BKPL MOPQ 1
		BKPL R 1
		Goto Ready
		
	Flash:
		TNT1 A 1 A_Light1
	Goto LightDone
		TNT1 B 1 A_Light1
		Goto LightDone

	Deathstate:
		TNT1 A 0 A_TakeInventory("DeathToken", 1)
		TNT1 A 0 A_Jump(255, "Deathstate1", "Deathstate2")
		stop
		
	Deathstate1:
		TNT1 A 0 Offset(0,32)
		TNT1 A 0 A_StopSound(CHAN_BODY)
		TNT1 A 0 A_StopSound(CHAN_WEAPON)
		TNT1 A 0 A_StopSound(CHAN_6)
		DED1 ABC 1
		DED1 DE 1
		DED1 FGHIJ 2
		DED1 J 5
		DED1 LMNOPQRSTU 1
		DED1 U -1
		stop
		
	Deathstate2:
		TNT1 A 0 Offset(0,32)
		TNT1 A 0 A_StopSound(CHAN_BODY)
		TNT1 A 0 A_StopSound(CHAN_WEAPON)
		TNT1 A 0 A_StopSound(CHAN_6)
		DED3 ABC 1
		DED3 DE 1
		DED3 FGHIJ 2
		DED3 J 5
		DED3 LMNOPQRSTU 1
		DED3 U -1
		stop
		
	DeathGibbed:
		TNT1 A 0
		TNT1 A 0 A_JumpIf(CallACS("Gibtoggle")==0, +2)
		TNT1 a 0 a_spawnitemex("gorecontroller")
		stop
		
	BLDeathstate:
		TNT1 A 0 A_TakeInventory("DeathToken", 1)
		TNT1 A 0 A_Jump(255, "BLDeathstate1", "BLDeathstate2")
		stop
		
	BLDeathstate1:
		TNT1 A 0 Offset(0,32)
		TNT1 A 0 A_StopSound(CHAN_BODY)
		TNT1 A 0 A_StopSound(CHAN_WEAPON)
		TNT1 A 0 A_StopSound(CHAN_6)
		DED2 ABC 1
		DED2 DE 1
		DED2 FGHIJ 2
		DED2 J 5
		DED2 LMNOPQRSTU 1
		DED2 U -1
		stop
		
	BLDeathstate2:
		TNT1 A 0 Offset(0,32)
		TNT1 A 0 A_StopSound(CHAN_BODY)
		TNT1 A 0 A_StopSound(CHAN_WEAPON)
		TNT1 A 0 A_StopSound(CHAN_6)
		DED4 ABC 1
		DED4 DE 1
		DED4 FGHIJ 2
		DED4 J 5
		DED4 LMNOPQRSTU 1
		DED4 U -1
		stop

/////////////
//64 STATES//		
/////////////

	64spawn:
		TNT1 A 0 A_JumpIf(CallACS("skintoggler")==0, "spawn")
		PL64 A 5
		loop
	64select:
		PLSG A 1 A_Raise
		Loop
	64deselect:
		PLSG A 1 A_Lower
		Loop
	64ready:
		PLSG ABC 2 A_WeaponReady
		Loop
	64Fire:
		TNT1 A 0 a_jumpif(callacs("BobToggle")==1, "64bobfire")
		TNT1 A 0 A_JumpIf(CallACS("recoiltoggle")==1, "64recoilfire")
		PLSG R 1 BRIGHT
		PLSF B 0 BRIGHT
		PLSF A 1 BRIGHT A_FirePlasma
		PLSF B 1 BRIGHT
		PLSF B 0 A_GunFlash
		PLSF CC 1 BRIGHT
		PLSG D 1 A_ReFire
		goto ready

	64bobFire:
		TNT1 A 0 A_JumpIf(CallACS("recoiltoggle")==1, "recoilbobfire")
		TNT1 A 0 A_WeaponReady(WRF_NoFire|WRF_NoSwitch)
		PLSG R 1 BRIGHT
		TNT1 A 0 A_WeaponReady(WRF_NoFire|WRF_NoSwitch)
		PLSF B 0 BRIGHT
		TNT1 A 0 A_WeaponReady(WRF_NoFire|WRF_NoSwitch)
		PLSF A 1 BRIGHT A_FirePlasma
		TNT1 A 0 A_WeaponReady(WRF_NoFire|WRF_NoSwitch)
		PLSF B 1 BRIGHT
		PLSF B 0 A_GunFlash
		TNT1 A 0 A_WeaponReady(WRF_NoFire|WRF_NoSwitch)
		PLSF C 2 BRIGHT
		TNT1 A 0 A_WeaponReady(WRF_NoFire|WRF_NoSwitch)
		PLSG D 1 A_ReFire
		goto ready
		

//////////////////
////BOB STATES////
//////////////////	


   	}
}
