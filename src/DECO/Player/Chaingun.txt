ACTOR Z86Chaingun : Chaingun replaces Chaingun
{
	Weapon.SlotNumber 4
	Weapon.BobStyle "Inverse"
	Weapon.BobSpeed 1.5
	Weapon.BobRangeX 1.00
	Weapon.BobRangeY 0.5
	decal bulletchip
	States
	{
	spawn:
		TNT1 A 0 nodelay A_JumpIf(CallACS("skintoggler")==1, "64spawn")
		MGUN a 5
		loop
    Ready:
		TNT1 A 0 A_JumpIf(CallACS("skintoggler")==1, "64Ready")
		PKCG A 1 A_WeaponReady
		Loop
    Deselect:
		TNT1 A 0 A_JumpIf(health <= 0, "DeadDeselect")
		TNT1 A 0 A_JumpIf(CallACS("skintoggler")==1, "64Deselect")
		TNT1 A 0 a_jumpif(callacs("skintoggler")==2, "bldeselect")
		CHGS PONMLKJIHGFEDCBA 1 A_Lower
		Loop
		
	BLDeselect:
		TNT1 A 0 A_JumpIf(health <= 0, "DeadBLDeselect")
		TNT1 A 0 A_JumpIf(CallACS("skintoggler")==1, "64Deselect")
		CHGS PONMLKJIHGFEDCBA 1 A_Lower
		Loop
		
	DeadDeselect:
		TNT1 A 0 A_JumpIf(CallACS("SkinToggler")==2, "DeadBLdeselect")
		TNT1 A 0 A_JumpIfInventory("GibToken", 1, "Deathgibbed")
		TNT1 A 0 A_JumpIfInventory("DeathToken", 1, "DeathState")
		PKCG A 1 A_Lower
		Loop
		
	DeadBLDeselect:
		TNT1 A 0 A_JumpIfInventory("GibToken", 1, "BLDeathgibbed")
		TNT1 A 0 A_JumpIfInventory("DeathToken", 1, "BLDeathState")
		PKCG A 1 A_Lower
		Loop
		
    Select:
		TNT1 A 0 a_jumpif(callacs("WeaponSoundToggle")==1, "silentselect")
		TNT1 A 0 a_jumpif(callacs("WeaponSoundToggle")==3, "silentselect")
		TNT1 A 0 A_JumpIf(CallACS("skintoggler")==1, "64Select")
		TNT1 A 0 A_Raise
		TNT1 A 0 A_PlaySound("weapons/chaingun/draw")
		CHGS ABCDEFGHIJKLMNOP 1 A_WeaponReady(WRF_NoFire)
		goto Ready
		
	SilentSelect:
		TNT1 A 0 a_jumpif(callacs("WeaponSoundToggle")==0, "sselect")
		TNT1 A 0 a_jumpif(callacs("WeaponSoundToggle")==2, "select")
		TNT1 A 0 A_JumpIf(CallACS("skintoggler")==1, "64Select")
		TNT1 A 0 A_Raise
		CHGS ABCDEFGHIJKLMNOP 1 A_WeaponReady(WRF_NoFire)
		goto Ready
		
	Fire:
		TNT1 A 0 A_JumpIf(CallACS("bobtoggle")==1, "bobFire")
		TNT1 A 0 A_JumpIf(CallACS("skintoggler")==1, "64fire")
		PKCG A 1 offset(0,38)
		{
			A_PlaySound("weapons/chngun", CHAN_WEAPON);
			A_GunFlash("Flash");
			A_FireBullets(5.6, 0, 1, 5, "AttackSmoothPuff");
		}
		TNT1 a 0 A_JumpIf(CallACS("CasingToggle")==0, +2)
		TNT1 a 0 A_spawnitemex ("bulletcasing", 50, 10, 15, random(0,1), random(3,5), random(4,7), 0)

			A_GunFlash("Flash2");
			A_FireBullets(5.6, 0, 1, 5, "AttackSmoothPuff");
		}
		TNT1 a 0 A_JumpIf(CallACS("CasingToggle")==0, +2)
		TNT1 a 0 A_spawnitemex ("bulletcasing", 50, 10, 15, random(0,1), random(3,5), random(4,7), 0)
		PKCG BCD 1
		PKCG A 1 
			{
			A_Weaponready;
			A_ReFire;
			}
		PKCG BB 1 
			{
			A_Weaponready;
			}
		PKCG CD 3 a_weaponready
		Goto Ready
		

	Flash:
		TNT1 A 0 A_JumpIf(CallACS("recoiltoggle")==1, "kickflash")	
		TNT1 A 0 A_JumpIf(CallACS("skintoggler")==1, "64kickflash")
			
		PKCF A 1 Bright A_Light2

		Goto LightDone

	Kickflash:
		PKCF A 1 Bright A_Light2

		PKCF B 1 Bright A_Light1
			tnt1 a 0 a_setpitch(pitch-0.1)
		
		TNT1 A 1 A_Light0

		Goto LightDone

		TNT1 A 1 A_Light0

		stop
		
	Deathstate1:

		stop
		
	Deathstate2:

	BLDeathstate:

	BLDeathstate1:

		CHGG A 1 A_PlaySound("weapons/chngun", CHAN_AUTO, 1, 0, ATTN_NORM)
		tnt1 a 0 A_JumpIf(CallACS("CasingToggle")==0, +2)
		TNT1 A 0 A_spawnitemex ("bulletcasing", 40, 0, 25, random(0,2), random(4,6), random(3,5), 0)

		CHGG B 1 Offset(-2, 34)

// A distinct high-rate alternative to the regular chaingun. It fires one
// lower-damage round every two tics, consuming ammunition rapidly in exchange
// for somewhat higher sustained damage than the regular chaingun.
ACTOR TuinMinigun : Z86Chaingun
{
	Tag "MINIGUN"
	Inventory.PickupMessage "You got the Minigun!"
	+INVENTORY.UNDROPPABLE
	States
	{
	Spawn:
		MNIG A -1
		Stop

	Ready:
		TNT1 A 0 A_JumpIfInventory("TuinMinigunInitialized", 1, "ReadyLoadedCheck")
		TNT1 A 0 A_GiveInventory("TuinMinigunInitialized", 1)
		TNT1 A 0 A_GiveInventory("TuinMinigunRounds", 100)
	ReadyLoadedCheck:
		TNT1 A 0 A_JumpIfInventory("TuinMinigunRounds", 1, "ReadyLoop")
		TNT1 A 0 A_JumpIfNoAmmo("Empty")
		Goto Reload
	ReadyLoop:
		VLCN A 1 A_WeaponReady(WRF_ALLOWRELOAD)
		Loop

	Select:
		VLCN A 1 A_Raise
		Loop

	Deselect:
		TNT1 A 0 A_StopSound(CHAN_WEAPON)
		VLCN A 1 A_Lower
		Loop

	Fire:
		TNT1 A 0 A_JumpIfInventory("TuinMinigunRounds", 1, +2)
		Goto Reload
		TNT1 A 0 A_JumpIfNoAmmo("Empty")
		TNT1 A 0 A_PlaySound("tuin/minigun/start", CHAN_AUTO)
		VLCN B 2
		VLCN CD 1
		VLCN ABCD 1
		TNT1 A 0 A_ReFire
		Goto WindDown
	Hold:
		TNT1 A 0 A_JumpIfInventory("TuinMinigunRounds", 1, +2)
		Goto Reload
		TNT1 A 0 A_JumpIfNoAmmo("EmptyWindDown")
		VLCS A 1
		{
			A_PlaySound("tuin/minigun/fire", CHAN_WEAPON);
			A_GunFlash("Flash");
			A_FireBullets(7.0, 0, 1, 3, "AttackSmoothPuff");
			A_TakeInventory("TuinMinigunRounds", 1);
		}
		VLCS B 1
		VLCS C 1
		{
			A_PlaySound("tuin/minigun/fire", CHAN_WEAPON);
			A_GunFlash("Flash");
			A_FireBullets(7.0, 0, 1, 3, "AttackSmoothPuff");
			A_TakeInventory("TuinMinigunRounds", 1);
		}
		VLCS D 1 A_ReFire

	AltFire:
		TNT1 A 0 A_JumpIfInventory("TuinMinigunRounds", 3, +2)
		Goto Reload
		TNT1 A 0 A_JumpIfInventory("Clip", 3, +2)
		Goto AltEmpty
		TNT1 A 0 A_PlaySound("tuin/minigun/start", CHAN_AUTO)
		VLCN BCD 2
		Goto AltHold
	AltHold:
		TNT1 A 0 A_JumpIfInventory("TuinMinigunRounds", 3, +2)
		Goto Reload
		TNT1 A 0 A_JumpIfInventory("Clip", 3, +2)
		Goto AltEmptyWindDown
		VLCN A 2 Bright
		{
			A_PlaySound("tuin/minigun/altfire", CHAN_WEAPON);
			A_GunFlash("Flash");
			A_FireCustomMissile("TuinMinigunExplosiveRound", 0, 0, 0, 0);
			A_TakeInventory("Clip", 3);
			A_TakeInventory("TuinMinigunRounds", 3);
		}
		VLCN BCD 3
		TNT1 A 0 A_ReFire("AltHold")
	WindDown:
		TNT1 A 0 A_PlaySound("tuin/minigun/stop", CHAN_AUTO)
		VLCN ABCD 2
		Goto Ready

	Reload:
		TNT1 A 0 A_JumpIfNoAmmo("Empty")
		TNT1 A 0 A_PlaySound("tuin/minigun/chain", CHAN_AUTO)
		RCLD ABCDEEE 1
		TNT1 A 0 A_PlaySound("tuin/minigun/reload", CHAN_AUTO)
		RCLD F 2
		NGLD GHIJKL 2
		TNT1 A 0 A_GiveInventory("TuinMinigunRounds", 100)
		Goto Ready

	EmptyWindDown:
		TNT1 A 0 A_PlaySound("tuin/minigun/stop", CHAN_AUTO)
	Empty:
		TNT1 A 0 A_PlaySound("weapons/pistol/dryf", CHAN_WEAPON)
		TNT1 A 0 A_Print("OUT OF BULLETS")
		TNT1 A 0 A_CheckReload
		VLCN A 3
		Goto ReadyLoop

	AltEmptyWindDown:
		TNT1 A 0 A_PlaySound("tuin/minigun/stop", CHAN_AUTO)
	AltEmpty:
		TNT1 A 0 A_PlaySound("weapons/pistol/dryf", CHAN_WEAPON)
		TNT1 A 0 A_Print("NEED 3 BULLETS")
		VLCN A 3
		Goto ReadyLoop

	Flash:
		CHAF A 1 Bright A_Light2
		CHAF B 1 Bright A_Light1
		Goto LightDone
	}
}
